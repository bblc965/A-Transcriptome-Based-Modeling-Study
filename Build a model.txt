# Confidence interval calculation tool function
binomial_ci <- function(x, n, conf.level = 0.95) {
  # Binomial distribution 95% confidence interval (Wilson method, more robust)
  if (n == 0) return(c(lower = NA, upper = NA))
  
  res <- binom.test(x, n, conf.level = conf.level)
  ci <- res$conf.int
  return(c(lower = ci[1], upper = ci[2]))
}
auc_ci <- function(roc_obj, conf.level = 0.95) {
  # The 95% confidence interval of AUC (obtained using the bootstrap method)
  ci_auc <- ci.auc(roc_obj, method = "bootstrap", boot.n = 2000, conf.level = conf.level)
  return(c(lower = ci_auc[1], upper = ci_auc[3]))
}

# Core indicator calculation function
calculate_metrics <- function(actual, predicted_prob, threshold = 0.5) {
  predicted_class <- ifelse(predicted_prob >= threshold, 1, 0)
  confusion <- table(Actual = actual, Predicted = predicted_class)
  
  TP <- ifelse("1" %in% rownames(confusion) & "1" %in% colnames(confusion), confusion["1", "1"], 0)
  TN <- ifelse("0" %in% rownames(confusion) & "0" %in% colnames(confusion), confusion["0", "0"], 0)
  FP <- ifelse("0" %in% rownames(confusion) & "1" %in% colnames(confusion), confusion["0", "1"], 0)
  FN <- ifelse("1" %in% rownames(confusion) & "0" %in% colnames(confusion), confusion["1", "0"], 0)
  
  # Point estimation
  sensitivity <- TP / (TP + FN)
  specificity <- TN / (TN + FP)
  accuracy <- (TP + TN) / sum(confusion)
  ppv <- TP / (TP + FP)
  npv <- TN / (TN + FN)
  
  # 95% confidence interval
  sens_ci <- binomial_ci(TP, TP + FN)
  spec_ci <- binomial_ci(TN, TN + FP)
  acc_ci  <- binomial_ci(TP + TN, sum(confusion))
  ppv_ci  <- binomial_ci(TP, TP + FP)
  npv_ci  <- binomial_ci(TN, TN + FN)
  
  return(list(
    confusion_matrix = confusion,
    sensitivity = sensitivity, sens_ci = sens_ci,
    specificity = specificity, spec_ci = spec_ci,
    accuracy = accuracy, acc_ci = acc_ci,
    ppv = ppv, ppv_ci = ppv_ci,
    npv = npv, npv_ci = npv_ci,
    TP = TP, TN = TN, FP = FP, FN = FN
  ))
}
# Output function
print_accuracy <- function(dataset_name, true_labels, predicted_probs, roc_obj) {
  best_threshold <- coords(roc_obj, "best", best.method = "youden")$threshold
  if (is.null(best_threshold) || length(best_threshold) == 0) best_threshold <- 0.5
  
  metrics <- calculate_metrics(true_labels, predicted_probs, best_threshold)
  auc_val <- auc(roc_obj)
  auc_ci_val <- auc_ci(roc_obj)
  
  # Formatted output: Value (95% CI: Lower limit - Upper limit)
  cat("\n==========", dataset_name, "==========\n")
  cat("cut-off :", round(best_threshold, 3), "\n")
  cat("AUC:", round(auc_val, 3), " (95%CI:", round(auc_ci_val[1],3), "-", round(auc_ci_val[2],3), ")\n")
  cat("Sensitivity:", round(metrics$sensitivity, 3), " (95%CI:", round(metrics$sens_ci[1],3), "-", round(metrics$sens_ci[2],3), ")\n")
  cat("Specificity:", round(metrics$specificity, 3), " (95%CI:", round(metrics$spec_ci[1],3), "-", round(metrics$spec_ci[2],3), ")\n")
  cat("Accuracy:", round(metrics$accuracy, 3), " (95%CI:", round(metrics$acc_ci[1],3), "-", round(metrics$acc_ci[2],3), ")\n")
  cat("PPV:", round(metrics$ppv, 3), " (95%CI:", round(metrics$ppv_ci[1],3), "-", round(metrics$ppv_ci[2],3), ")\n")
  cat("NPV:", round(metrics$npv, 3), " (95%CI:", round(metrics$npv_ci[1],3), "-", round(metrics$npv_ci[2],3), ")\n")
  cat("Confusion matrix:\n")
  print(metrics$confusion_matrix)
}

# Customize the Bootstrap calibration function 
calibrate_glmnet <- function(pred_prob, y_true, n_boot = 1000, n_seq = 200) {
  # Generate a sequence of predicted probabilities ranging from 0 to 1
  pred_seq <- seq(0, 1, length.out = n_seq)
  # Original observation probability (loess smoothing, without line)
  loess_orig <- loess(y_true ~ pred_prob, span = 0.75, degree = 1, control = loess.control(surface = "direct"))
  orig_smooth <- predict(loess_orig, newdata = data.frame(pred_prob = pred_seq))
  # Limit the original probability to be within the range of 0 to 1 (to prevent the graph from overflowing)
  orig_smooth[orig_smooth < 0] <- 0
  orig_smooth[orig_smooth > 1] <- 1
  # Bootstrap correction: Iteratively calculate the corrected smoothed probability
  boot_smooth <- matrix(NA, nrow = n_boot, ncol = n_seq)
  for (b in 1:n_boot) {
    # Sampling with replacement
    samp_idx <- sample(1:length(y_true), replace = TRUE)
    p_samp <- pred_prob[samp_idx]
    y_samp <- y_true[samp_idx]
    # Perform loess smoothing on the sample data
    loess_boot <- loess(y_samp ~ p_samp, span = 0.75, degree = 1,control = loess.control(surface = "direct"))
    boot_smooth[b, ] <- predict(loess_boot, newdata = data.frame(p_samp = pred_seq))
  }
  # Calculate the probability after Bootstrap correction
  corr_smooth <- 2 * orig_smooth - colMeans(boot_smooth, na.rm = TRUE)
  # Limit the correction probability to be within the range of 0 to 1.
  corr_smooth[corr_smooth < 0] <- 0
  corr_smooth[corr_smooth > 1] <- 1
  # Perform a mild secondary smoothing on the original curve
  loess_orig2 <- loess(orig_smooth ~ pred_seq, span = 0.75, degree = 1)
  orig_smooth <- predict(loess_orig2, newdata = data.frame(pred_seq = pred_seq))
  orig_smooth[orig_smooth < 0] <- 0
  orig_smooth[orig_smooth > 1] <- 1
  # Perform a mild second-order smoothing on the calibration curve
  loess_corr2 <- loess(corr_smooth ~ pred_seq, span = 0.75, degree = 1)
  corr_smooth <- predict(loess_corr2, newdata = data.frame(pred_seq = pred_seq))
  corr_smooth[corr_smooth < 0] <- 0
  corr_smooth[corr_smooth > 1] <- 1
  
  return(data.frame(pred_seq, orig_smooth, corr_smooth))
}
