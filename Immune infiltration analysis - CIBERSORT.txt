# Loading LM22 reference set
load("LM22.rda")
sig_matrix <- LM22
# Read in the FPKM expression matrix
mixture_file <- read.csv("gene_fpkm.csv", row.names = 1, check.names = F)
common_genes <- intersect(rownames(sig_matrix), rownames(mixture_file))
sig_matrix <- sig_matrix[common_genes, ]
mixture_file <- mixture_file[common_genes, ]
write.table(mixture_file, "mixture_temp.txt", sep = "\t", row.names = TRUE, col.names = TRUE, quote = FALSE)
# Run CIBERSORT
res_cibersort <- cibersort(sig_matrix = sig_matrix, 
                           mixture_file = "mixture_temp.txt", 
                           perm = 100, 
                           QN = TRUE, 
                           maxSize = 2000)
# Extract 22 types of immune cells
rownames(res_cibersort) <- colnames(mixture_file) 
res_cibersort <- res_cibersort[,1:22]
# Remove all cells with zero values
ciber.res <- res_cibersort[,colSums(res_cibersort) > 0]
ciber.res <- as.data.frame(ciber.res)
write.table(ciber.res, "ciber.res.txt", sep = "\t", row.names = TRUE, col.names = TRUE, quote = FALSE)

#Box plot
library(ggpubr)
library(ggsci)
library(rstatix)
library(tidyverse)
# Read in data
ciber.res <- read.table("ciber.res.txt", sep="\t", row.names=1, header=T)
group <- read.csv("Sample grouping.csv", row.names=1, stringsAsFactors=F)
colnames(group) <- "group"
ciber.res$group <- group[rownames(ciber.res), "group"]
# Sort out the data
plot_data <- ciber.res %>%
  rownames_to_column("sample") %>%
  pivot_longer(cols = -c(sample,group), names_to="CellType", values_to="Fraction")

# Color scheme
plot_data$group <- factor(plot_data$group, levels = c("Healthy","Common Infection","Sepsis"))
my_col <- c("Healthy"="green", "Common Infection"="blue", "Sepsis"="red")

# Pairwise comparison group 
comp <- list(c("Healthy","Common Infection"),
             c("Healthy","Sepsis"),
             c("Common Infection","Sepsis"))
#KW test
kw_results <- plot_data %>%
  group_by(CellType) %>%
  summarise(
    KW_p_value = kruskal.test(Fraction ~ group)$p.value,
    .groups = "drop"
  ) %>%
  mutate(
    KW_p_signif = case_when(
      KW_p_value < 0.0001 ~ "****",
      KW_p_value < 0.001 ~ "***",
      KW_p_value < 0.01 ~ "**",
      KW_p_value < 0.05 ~ "*",
      TRUE ~ "ns"
    )
  )

# Calculate the P-values for pairwise Wilcoxon tests (post-hoc tests)
wilcox_results <- plot_data %>%
  group_by(CellType) %>%
  pairwise_wilcox_test(
    Fraction ~ group,
    p.adjust.method = "BH"  
  ) %>%
  select(CellType, group1, group2, p, p.adj, p.adj.signif)

# Save as CSV file
write.csv(kw_results, "KW test result - Differences between immune cell groups.csv", row.names = FALSE)
write.csv(wilcox_results, "Pairwise Wilcoxon test results_ Differences between immune cell groups.csv", row.names = FALSE)

ggboxplot(plot_data, x="CellType", y="Fraction", fill="group", palette = my_col) +
  
  # KW test (retained)
  stat_compare_means(aes(group=group), method="kruskal.test", label="p.signif") +
  

  stat_compare_means(
    aes(group=group),
    method = "wilcox.test",
    comparisons = comp,    
    label = "p.signif",    
    hide.ns = T           
  ) +
  
  theme(axis.text.x = element_text(angle=45, hjust=1))
ggsave("Comparison chart of immune infiltration in the three groups.pdf", width=6, height=6)

#Immune cell-related forest plot
library(tidyverse)
library(ggplot2)
expr <- read.csv("gene_fpkm.csv", row.names = 1, check.names = F, stringsAsFactors = F)
expr <- as.data.frame(t(expr))
expr <- as.data.frame(lapply(expr, as.numeric))  
rownames(expr) <- colnames(read.csv("gene_fpkm.csv", row.names = 1, check.names = F))  
# Read in the results of immune infiltration
ciber.res <- read.table("ciber.res.txt", sep="\t", row.names=1, header=T, check.names=F)
ciber.res[] <- lapply(ciber.res, as.numeric) 
# Sample alignment (ensuring that the sample order is completely consistent)
common_samples <- intersect(rownames(expr), rownames(ciber.res))
expr <- expr[common_samples, ]
ciber.res <- ciber.res[common_samples, ]

gene_name <- "***"# Specified gene
#Extract gene expression levels
gene_data <- expr[[gene_name]]
# Calculate correlation (with fault tolerance)
cor_res <- map_dfr(colnames(ciber.res), function(cell){
  ct <- cor.test(gene_data, ciber.res[[cell]], method = "spearman")
  data.frame(Cell = cell, R = ct$estimate, P = ct$p.value)
})
# Sort (by correlation coefficient from low to high)
cor_res <- cor_res %>% arrange(R)
cor_res$Cell <- factor(cor_res$Cell, levels = cor_res$Cell)
# Drawing
ggplot(cor_res, aes(x=R, y=Cell)) +
  geom_vline(xintercept=0, linetype="dashed", color="gray") +
  
  geom_errorbar(
    aes(xmin=pmin(R,0), xmax=pmax(R,0)),
    orientation="y", width=0.2, linewidth=0.8, color="black"
  ) +
  
  geom_point(
    aes(fill=P<0.05),
    size=5, shape=21, color="black", stroke=0.8
  ) +
  
  geom_text(
    aes(x=0.85, label=ifelse(P<0.001, "<0.001", round(P,3)), color=P<0.05),
    hjust=0, size=3.8
  ) +
  
  scale_fill_manual(values=c("FALSE"="blue","TRUE"="red"), guide="none") +
  scale_color_manual(values=c("FALSE"="black","TRUE"="red"), guide="none") +
  
  theme(panel.background = element_rect(fill = "gray100"),
        panel.grid.major = element_line(color = "gray90"),
        panel.grid.minor = element_line(color = "gray90"),
        
        plot.margin = unit(c(0.5, 2, 0.5, 0.5), "cm")) +
  labs(title=gene_name, x="Correlation coefficient", y="") +
  
  coord_cartesian(xlim = c(-0.6, 0.7), clip = "off")

# Save as PDF
ggsave(paste0(gene_name, "_Correlation graph.pdf"), width=6, height=6, dpi=300)
